def _download_article_with_timeout(url: str, timeout: int = 15) -> Optional[Dict]:
    """Download article using newspaper3k with timeout."""
    def _download():
        article = NewspaperArticle(url)
        article.download()
        article.parse()
        return {
            'title': article.title or "",
            'text': article.text or "",
            'authors': article.authors or [],
            'publish_date': article.publish_date
        }
    try:
        with ThreadPoolExecutor(max_workers=1) as executor:
            future = executor.submit(_download)
            return future.result(timeout=timeout)
    except:
        return None
class SerperClient:
    """Search via Serper.dev (Google Search API)."""
    
    def __init__(self):
        self.api_key = os.environ.get("SERPER_API_KEY")
        self.base_url = "https://google.serper.dev"
    
    def search(self, query: str, num_results: int = 20) -> List[Dict]:
        if not self.api_key:
            return []
        try:
            response = requests.post(
                f"{self.base_url}/search",
                headers={"X-API-KEY": self.api_key, "Content-Type": "application/json"},
                json={"q": query, "num": num_results},
                timeout=15
            )
            data = response.json()
            return [
                {"title": r.get("title", ""), "url": r.get("link", ""), 
                 "snippet": r.get("snippet", ""), "date": r.get("date", "")}
                for r in data.get("organic", [])
            ]
        except Exception as e:
            print(f"Serper error: {e}")
            return []
    
    def search_news(self, query: str, num_results: int = 20) -> List[Dict]:
        if not self.api_key:
            return []
        try:
            response = requests.post(
                f"{self.base_url}/news",
                headers={"X-API-KEY": self.api_key, "Content-Type": "application/json"},
                json={"q": query, "num": num_results},
                timeout=15
            )
            data = response.json()
            return [
                {"title": r.get("title", ""), "url": r.get("link", ""),
                 "snippet": r.get("snippet", ""), "date": r.get("date", ""),
                 "source": r.get("source", "")}
                for r in data.get("news", [])
            ]
        except Exception as e:
            print(f"Serper news error: {e}")
            return []
class GDELTClient:
    """Free GDELT API for global news search."""
    
    def __init__(self):
        self.base_url = "https://api.gdeltproject.org/api/v2/doc/doc"
    
    def search(self, query: str, years_back: int = 2, max_results: int = 75) -> List[Dict]:
        import time
        timespan = f"{years_back}y" if years_back > 0 else None
        full_query = f'{query} sourcecountry:US sourcelang:eng'
        
        params = {
            "query": full_query,
            "mode": "artlist",
            "maxrecords": min(max_results, 75),
            "format": "json",
            "sort": "datedesc"
        }
        if timespan:
            params["timespan"] = timespan
        
        for attempt in range(3):
            try:
                response = requests.get(self.base_url, params=params, timeout=30)
                if response.status_code == 429:
                    time.sleep(2 ** attempt)
                    continue
                if response.status_code == 200:
                    data = response.json()
                    return [
                        {"url": a.get("url", ""), "title": a.get("title", ""),
                         "source": a.get("domain", ""), "date": a.get("seendate", "")}
                        for a in data.get("articles", [])
                    ]
            except:
                time.sleep(2)
        return []

import os
from typing import List, Optional
class LLMFactory:
    @staticmethod
    def get_available_models() -> List[str]:
        models = []
        if os.environ.get("GOOGLE_API_KEY"):
            models.append("gemini-1.5-pro")
        if os.environ.get("ANTHROPIC_API_KEY"):
            models.append("claude-3-5-sonnet-20241022")
        if os.environ.get("OPENAI_API_KEY"):
            models.append("gpt-4o")
        return models
    
    @staticmethod
    def get_model(model_name: str):
        if "gemini" in model_name:
            return GeminiModel()
        elif "claude" in model_name:
            return ClaudeModel()
        elif "gpt" in model_name:
            return OpenAIModel()
        raise ValueError(f"Unknown model: {model_name}")
class GeminiModel:
    def generate(self, system_prompt: str, user_prompt: str, max_tokens: int = 4000) -> str:
        import google.generativeai as genai
        genai.configure(api_key=os.environ.get("GOOGLE_API_KEY"))
        model = genai.GenerativeModel('gemini-1.5-pro')
        response = model.generate_content(
            f"{system_prompt}\n\n{user_prompt}",
            generation_config=genai.GenerationConfig(max_output_tokens=max_tokens)
        )
        return response.text
class ClaudeModel:
    def generate(self, system_prompt: str, user_prompt: str, max_tokens: int = 4000) -> str:
        from anthropic import Anthropic
        client = Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=max_tokens,
            system=system_prompt,
            messages=[{"role": "user", "content": user_prompt}]
        )
        return response.content[0].text
class OpenAIModel:
    def generate(self, system_prompt: str, user_prompt: str, max_tokens: int = 4000) -> str:
        from openai import OpenAI
        client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
        response = client.chat.completions.create(
            model="gpt-4o",
            max_tokens=max_tokens,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ]
        )
        return response.choices[0].message.content
