// Parse CivicTracker timestamp format: "Jan 20, 2026 • 1:53 AM"
function parseTimestamp(timestampStr: string): Date {
  const cleaned = timestampStr.replace('•', '').trim();
  
  const months: Record<string, number> = {
    'Jan': 0, 'Feb': 1, 'Mar': 2, 'Apr': 3, 'May': 4, 'Jun': 5,
    'Jul': 6, 'Aug': 7, 'Sep': 8, 'Oct': 9, 'Nov': 10, 'Dec': 11
  };
  
  const match = cleaned.match(/([A-Z][a-z]{2})\s+(\d{1,2}),\s+(\d{4})\s+(\d{1,2}):(\d{2})\s*(AM|PM)/i);
  
  if (match) {
    const [, monthStr, day, year, hour, minute, ampm] = match;
    let hours = parseInt(hour);
    
    // Convert to 24-hour format
    if (ampm.toUpperCase() === 'PM' && hours !== 12) hours += 12;
    if (ampm.toUpperCase() === 'AM' && hours === 12) hours = 0;
    
    return new Date(
      parseInt(year),
      months[monthStr],
      parseInt(day),
      hours,
      parseInt(minute)
    );
  }
  
  // Fallback to current time
  console.warn(`Could not parse timestamp: ${timestampStr}`);
  return new Date();
}

  const posts: TruthSocialPostData[] = [];

    console.log('Starting CivicTracker backup scraper...');

      args: ['--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage']

    });
    const page = await context.newPage();
    
    await page.goto('https://civictracker.us/executive/member/?uuid=3094abf7-4a95-4b8d-8c8d-af7d1c3747a1', {
      waitUntil: 'networkidle',
      timeout: 30000
    });
    
    // Wait for dynamic content to load
    await page.waitForTimeout(3000);
    
    // Extract posts from the page
    const pageContent = await page.evaluate(() => {
      const fullText = document.body.innerText || '';
      const results: Array<{content: string, timestamp: string, imageUrl: string | null}> = [];
      
      // CivicTracker timestamp pattern
      const timestampPattern = /([A-Z][a-z]{2}\s+\d{1,2},\s+\d{4}\s*•\s*\d{1,2}:\d{2}\s*[AP]M)/g;
      
      // Find all timestamps
      const timestamps: Array<{match: string, index: number}> = [];
      let m;
      while ((m = timestampPattern.exec(fullText)) !== null) {
        timestamps.push({ match: m[0], index: m.index });
      }
      
      // Get archive.org image URLs for post IDs
      const images = Array.from(document.querySelectorAll('img[src*="archive.org"]')) as HTMLImageElement[];
      const imageUrls = images.map(img => img.src);
      
      // For each timestamp, extract content block before it
      for (let i = 0; i < timestamps.length; i++) {
        const ts = timestamps[i];
        const prevEnd = i > 0 ? timestamps[i-1].index + timestamps[i-1].match.length : 0;
        
        let contentBlock = fullText.substring(prevEnd, ts.index).trim();
        
        // Clean up boilerplate
        contentBlock = contentBlock
          .replace(/Donald J\. Trump\s*President of the United States/g, '')
          .replace(/Loading executive member…/g, '')
          .replace(/Your browser does not support the video tag\./g, '')
          .replace(/Post media/g, '')
          .replace(/\n{3,}/g, '\n\n')
          .trim();
        
        // Get the last substantial line (actual post content)
        const lines = contentBlock.split('\n').filter(l => l.trim().length > 10);
        const postContent = lines[lines.length - 1]?.trim() || '';
        
        // Filter out placeholder/navigation text
        if (postContent.length > 15 && 
            !postContent.includes('CivicTracker') && 
            !postContent.includes('Sign In') &&
            !postContent.includes('Upgrade to Pro') &&
            !postContent.toLowerCase().includes('click to view')) {
          
          const imageUrl = imageUrls[i] || null;
          
          results.push({
            content: postContent,
            timestamp: ts.match,
            imageUrl: imageUrl
          });
        }
      }
      
      return results;
    });
    
    console.log(`CivicTracker: Found ${pageContent.length} potential posts`);
    
    for (const item of pageContent) {
      // Additional content cleanup
      let content = item.content
        .replace(/Donald J\. Trump\s*President of the United States/g, '')
        .replace(/Post media/g, '')
        .replace(/Your browser does not support the video tag\./g, '')
        .trim();
      
      // Get last substantial paragraph
      const paragraphs = content.split(/\n/).filter(p => p.trim().length > 10);
      const lastParagraph = paragraphs[paragraphs.length - 1]?.trim();
      
      if (!lastParagraph || lastParagraph.length < 10) continue;
      if (lastParagraph.toLowerCase().includes('click to view')) continue;
      
      // Skip if we've already seen this content in this scrape
      if (seenContent.has(lastParagraph.substring(0, 50))) continue;
      seenContent.add(lastParagraph.substring(0, 50));
      
      const parsedTime = parseTimestamp(item.timestamp);
      
      // Generate post ID from archive.org image URL or content hash
      let postId = '';
      if (item.imageUrl) {
        const match = item.imageUrl.match(/\/(\d{15,20})_media/);
        if (match) {
          postId = match[1];
        }
      }
      
      // Fallback: generate from content hash with ct_ prefix
      if (!postId) {
        const contentHash = lastParagraph.split('').reduce((a, b) => {
          a = ((a << 5) - a) + b.charCodeAt(0);
          return a & a;
        }, 0);
        postId = `ct_${Math.abs(contentHash)}`;
      }

        postId,
        content: lastParagraph,
        postedAt: parsedTime,
        truthSocialUrl: postId.startsWith('ct_') 
          ? 'https://truthsocial.com/@realDonaldTrump'
          : `https://truthsocial.com/@realDonaldTrump/posts/${postId}`,
      });
    }
    
    await browser.close();
    activeBrowser = null;
    
    console.log(`CivicTracker backup: Scraped ${posts.length} posts`);
    return posts.slice(0, 20);

    console.error('Backup scraper error:', error);
    if (activeBrowser) {
      await activeBrowser.close();
      activeBrowser = null;
    }
    return [];
  }

import { scrapeTruthSocialPrimary } from './truthSocialScraper';
import { scrapeTruthSocialBackup } from './truthSocialBackupScraper';
import { IStorage } from './storage';
export async function scrapeAllTruthSocial(storage: IStorage): Promise<number> {
  let addedCount = 0;
  let primaryCount = 0;
  let backupCount = 0;
  
  // 1. Scrape primary source (TrumpsTruth.org)
  console.log('Scraping primary source (TrumpsTruth.org)...');
  const primaryPosts = await scrapeTruthSocialPrimary();
  
  for (const post of primaryPosts) {
    try {
      // Check if post already exists by post ID
      const existing = await storage.getTruthSocialPost(post.postId);
      if (!existing) {
        await storage.createTruthSocialPost(post);
        primaryCount++;
        addedCount++;
      }
    } catch (error) {
      console.error(`Error saving primary post ${post.postId}:`, error);
    }
  }
  
  console.log(`Primary source: Added ${primaryCount} new posts`);
  
  // 2. Scrape backup source (CivicTracker) for any missing posts
  console.log('Scraping backup source (CivicTracker)...');
  const backupPosts = await scrapeTruthSocialBackup();
  
  for (const post of backupPosts) {
    try {
      // Content-based deduplication: check first 50 characters
      const existingByContent = await storage.getTruthSocialPostByContentPrefix(
        post.content.substring(0, 50)
      );
      
      if (existingByContent) {
        console.log(`Skipping duplicate backup post (content match): ${post.content.substring(0, 50)}...`);
        continue;
      }
      
      // Also check by post ID
      const existingById = await storage.getTruthSocialPost(post.postId);
      if (!existingById) {
        await storage.createTruthSocialPost(post);
        backupCount++;
        addedCount++;
      }
    } catch (error) {
      console.error(`Error saving backup post ${post.postId}:`, error);
    }
  }
  
  console.log(`Backup source: Added ${backupCount} new posts`);
  console.log(`Added ${addedCount} new Truth Social posts total (${primaryCount} primary, ${backupCount} backup)`);
  
  return addedCount;
}

              </a>
            </div>
            <p className="text-sm">{post.content}</p>
          </Card>
        ))}
      </div>
    </div>
  );
}

Key Points Summary
Dual-source scraping: Primary (TrumpsTruth.org via Cheerio) + Backup (CivicTracker via Playwright)
Content-based deduplication: Compare first 50 characters to prevent duplicate posts from different sources
Placeholder filtering: Skip any posts containing "click to view"
Post ID prefixing: Backup posts use ct_ prefix to identify source
Schema field: Use postedAt not publishedAt for the timestamp column
Re-entrancy guard: Prevent overlapping scrape cycles with a flag
Scheduled scraping: Run every 10 minutes with initial scrape on startup
